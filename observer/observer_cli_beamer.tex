\begin{frame}{先接上 Observer UI 那条主线}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \begin{alertblock}{UI 版已经建立的直觉}
      Erlang 最关键的运行时对象是 \keypoint{Process}：它有 PID、mailbox、state、heap、stack，也可能拥有 ETS / DETS 资源。
    \end{alertblock}
    \begin{block}{这一集不换主题}
      \begin{itemize}
        \item 仍然观察同一个 BEAM node
        \item 仍然从 process、application、ETS 入手
        \item 只是把图形化操作换成终端操作
        \item 重点从结构展示转向指标排查
      \end{itemize}
    \end{block}
    \column{0.42\textwidth}
    \begin{exampleblock}{一句话定位}
      \centering
      \Large
      Observer UI 适合\par
      \good{看懂 OTP 结构}\par\vspace{0.25cm}
      observer\_cli 适合\par
      \good{在线上快速定位异常}
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{Observer UI 和 observer\_cli 的关系}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{Observer UI}
      \begin{itemize}
        \item 图形界面，适合本地开发和教学演示
        \item Applications 页面能展开 supervisor tree
        \item 可以沿着 application 找到 worker process
        \item 适合建立 OTP 结构直觉
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{exampleblock}{observer\_cli}
      \begin{itemize}
        \item 终端界面，适合 SSH、容器、无桌面环境
        \item Home 页面直接展示节点级指标
        \item Process 列表适合按 memory / reductions / MsgQ 排序
        \item ETS、System、Network 更偏运行时排查
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.25cm}
  \centering
  \good{两者不是替代关系，而是同一套 BEAM 可观测能力的两种入口。}
\end{frame}

\begin{frame}{终端场景下，问题会换一种问法}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{block}{UI 版更常问}
      \begin{itemize}
        \item 这个 application 的 root supervisor 是谁？
        \item supervisor 下面有哪些 worker？
        \item gen\_server 的 state 当前是什么？
        \item OTP 层级和代码如何对应？
      \end{itemize}
    \end{block}
    \column{0.46\textwidth}
    \begin{alertblock}{CLI 版更常问}
      \begin{itemize}
        \item 哪个 process 最占内存？
        \item 谁的 reductions 增长最快？
        \item 哪个 mailbox 正在堆积？
        \item 哪张 ETS 表占用异常？
      \end{itemize}
    \end{alertblock}
  \end{columns}
  \vspace{0.25cm}
  \centering
  \warn{所以 CLI 版的主线不是“展开结构”，而是“按指标缩小范围”。}
\end{frame}

\begin{frame}{Home 页面不是截图目录，而是排查入口}
  \begin{columns}[T,onlytextwidth]

    \begin{block}{从全局指标开始}
      \begin{itemize}
        \item process count 是否异常增长
        \item total memory 是否持续上升
        \item scheduler run queue 是否过高
        \item ETS / binary memory 是否可疑
      \end{itemize}
    \end{block}
    \column{0.46\textwidth}
    \begin{alertblock}{下一步缩小范围}
      \begin{itemize}
        \item 内存异常：看 process / ETS 排名
        \item CPU 异常：看 reductions 排名
        \item 延迟异常：看 message queue 排名
        \item 连接异常：看 port / network 信息
      \end{itemize}
    \end{alertblock}
  \end{columns}
  \vspace{0.25cm}
  \centering
  \good{CLI 的优势，是把“哪里不对劲”快速变成“先看哪一组对象”。}

\begin{frame}{ETS 不是旁支，它常常是 process 问题的一部分}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{block}{为什么要看 owner？}
      \begin{itemize}
        \item ETS table 通常由某个 process 拥有
        \item 表内对象增长，可能来自 owner 的业务逻辑
        \item owner 退出时，表生命周期也要考虑
        \item \code{data\_cache} 这类 worker 常见于缓存场景
      \end{itemize}
    \end{block}
    \column{0.46\textwidth}
    \begin{alertblock}{排查路径}
      \centering
      \code{ETS memory 异常}\par
      $\downarrow$\par
      \code{找到 table owner}\par
      $\downarrow$\par
      \code{回到 process 详情}\par
      $\downarrow$\par
      \code{确认 state / mailbox / current function}
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{observer\_cli 背后不是魔法}
  \begin{block}{它本质上依赖 BEAM 可观测 API}
    \begin{itemize}
      \item \code{erlang:system\_info/1}
      \item \code{erlang:memory/0}
      \item \code{application:which\_applications/0}
      \item \code{erlang:processes/0}
      \item \code{erlang:process\_info/2}
      \item \code{ets:all/0} / \code{ets:info/1}
    \end{itemize}
  \end{block}
  \begin{exampleblock}{demo 中的辅助脚本}
    \code{full\_otp\_app/scripts/observer\_cli\_snapshot.escript} 直接调用这些 API 采集一次快照。
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{远程节点参数：node name，不是 OS PID}

\end{lstlisting}
    \end{block}
    \column{0.46\textwidth}
    \begin{block}{脚本远程采集形式}

  \end{alertblock}
\end{frame}

% ── Teaching flow ──────────────────────────────────────────────────
\begin{frame}{课堂演示顺序}
  \small
  \begin{enumerate}
    \item 回顾 UI 版结论：Process 是 BEAM 运行时观察的核心对象
    \item 启动 \code{full\_otp\_app}，执行 \code{observer\_cli:start().}
    \item 看 Home 页面，说明这是终端版运行时仪表盘
    \item 按 \code{A} 看 application 聚合指标，并说明它不展开 supervisor tree
    \item 用 \code{supervisor:which\_children/1} 补充查看监督树
    \item 回到 process 列表，按 memory / reductions / message queue 排序
    \item 输入进程行号查看进程详情，把指标落回 process
    \item 按 \code{E} 查看 ETS 页面，从 table owner 回到 process
    \item 最后引出 Erlang runtime introspection API 和 snapshot 脚本
  \end{enumerate}
\end{frame}

\begin{frame}{什么时候用 observer\_cli？}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{推荐场景}
      \begin{itemize}
        \item SSH 到服务器
        \item 容器内排查
        \item 没有图形界面
        \item 想快速按指标排序
        \item 看 ETS / process / memory
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{alertblock}{不适合独立承担的场景}
      \begin{itemize}
        \item 教初学者第一次理解 supervisor tree
        \item 希望像 GUI 一样展开 application
        \item 需要大量截图标注 OTP 结构
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{小结：CLI 版补上的是线上观察能力}
  \begin{block}{本集结论}
    \begin{itemize}
      \item \code{observer\_cli} 观察的仍然是 UI 版那套 BEAM 运行时对象
      \item Home 页面先判断节点级运行状态
      \item App 页面看 application 聚合指标，不展开监督树
      \item Process 列表通过 memory / reductions / MsgQ 缩小范围
      \item ETS 页面从表指标继续追到 owner process
      \item 背后依赖 Erlang runtime introspection API
    \end{itemize}
  \end{block}
  \begin{exampleblock}{和 Observer UI 配合}
    \centering
    \Large
    UI 版讲清楚\good{系统结构}。\par\vspace{0.2cm}
    CLI 版训练\good{终端排查路径}。
  \end{exampleblock}
\end{frame}
