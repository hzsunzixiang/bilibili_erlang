% ── Importance ─────────────────────────────────────────────────────
\begin{frame}{为什么 Observer 值得先讲？}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \begin{alertblock}{这不是普通监控面板}
      很多语言生态也有 metrics、trace、profiler，但通常看到的是 OS 线程、HTTP 请求、堆内存或业务指标。
      \keypoint{Observer 直接站在 BEAM 运行时内部看系统。}
    \end{alertblock}
    \begin{block}{它能看到 Erlang 的核心建模单位}
      \begin{itemize}
        \item Erlang process 是语言级轻量进程
        \item 概念上接近 Go goroutine，但属于 BEAM VM
        \item 每个 process 有 mailbox、state、links、reductions
        \item Observer 把这些运行时对象直接展示出来
      \end{itemize}
    \end{block}
    \column{0.44\textwidth}
    \begin{exampleblock}{一句话定位}
      \centering
      \Large
      Erlang 的并发模型\par
      不是“猜出来”的，\par
      是可以\good{直接看见}的。
    \end{exampleblock}
  \end{columns}
\end{frame}

\begin{frame}{和常见语言工具相比，Observer 特别在哪里？}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{常见产品/语言工具更多看到}
      \begin{itemize}
        \item OS process / OS thread
        \item request latency / QPS
        \item heap / GC / CPU profile
        \item log / trace / span
        \item 外部采集后的指标视角
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{exampleblock}{Observer 直接看到}
      \begin{itemize}
        \item BEAM node 内部资源
        \item application 列表
        \item supervisor tree
        \item 每个 Erlang process
        \item mailbox / state / stack trace
        \item ETS / DETS 表
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.25cm}
  \centering
  \warn{关键差异：Observer 的观察粒度直接下沉到 Erlang process 级别。}
\end{frame}

\begin{frame}{Erlang process：类似 goroutine，但可观察性更内置}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{类比 Go goroutine}
      \begin{itemize}
        \item 都是轻量级并发单元
        \item 都不是手工创建 OS thread
        \item 都由运行时调度
        \item 都适合海量并发建模
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \begin{exampleblock}{Erlang + Observer 的特殊组合}
      \begin{itemize}
        \item process 有 mailbox
        \item process 有独立状态
        \item process 之间 link / monitor
        \item supervisor 管理 process 生命周期
        \item Observer 把这些关系可视化
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.25cm}
  \centering
  \good{这就是为什么讲 OTP 之前，先讲 Observer。}
\end{frame}

% ── Demo flow ──────────────────────────────────────────────────────
\begin{frame}{推荐课堂演示顺序：Applications 作为主线}
  \small
  \begin{enumerate}
    \item 启动 \code{rebar3 shell --sname application\_demo}
    \item 执行 \code{observer:start().}
    \item 先看 \code{System}，说明 Observer 是 BEAM node 内置观察入口
    \item 切到 \code{Applications}，强调这是本讲重点
    \item 选中 \code{full\_otp\_app}，解释 application 是 OTP 顶层组织单位
    \item 展开 supervision tree，对照 TikZ 层级图
    \item 双击 \code{kv\_server}，说明 gen\_server 是 process
    \item 查看 \code{State}，执行 \code{kv\_server:put/2} 后观察变化
    \item 双击 \code{traffic\_light}，说明 gen\_statem 当前状态可观察
    \item 用 \code{Messages} / \code{Stack Trace} / \code{Table Viewer} 补充排查视角
  \end{enumerate}
\end{frame}

\begin{frame}{小结：Observer UI 的真正意义}
  \begin{block}{它让 Erlang/OTP 的核心抽象可见}
    \begin{itemize}
      \item application 不只是配置名，可以在 UI 中选中
      \item supervisor 不只是策略，它本身是 process
      \item gen\_server / gen\_statem 不只是 callback，它们是有状态、有 mailbox 的进程
      \item Erlang process 作为并发基本单位，可以被直接观察
    \end{itemize}
  \end{block}
  \begin{exampleblock}{最重要的一句话}
    \centering
    \Large
    Observer 把 OTP 从“代码里的抽象”变成了\good{运行时的一览无余}。
  \end{exampleblock}
\end{frame}
