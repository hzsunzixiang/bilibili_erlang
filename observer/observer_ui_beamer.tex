\begin{frame}{为什么先讲 Observer UI？}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \begin{alertblock}{直接从 callback 开始，容易学成 API 清单}
      \code{application}、\code{supervisor}、\code{gen\_server}、\code{gen\_statem} 都是运行时结构。
      只看代码，不容易建立整体图像。
    \end{alertblock}
    \begin{block}{Observer 的价值}
      \begin{itemize}
        \item 看见当前 BEAM 节点
        \item 看见 application 列表
        \item 展开 supervisor tree
        \item 双击 process 查看状态、消息、调用栈
      \end{itemize}
    \end{block}
    \column{0.44\textwidth}
    \begin{exampleblock}{学习路线}
      \centering
      \code{启动 OTP 应用}\par
      $\downarrow$\par
      \code{Observer 看 application}\par
      $\downarrow$\par
      \code{展开 supervisor tree}\par
      $\downarrow$\par
      \code{回到代码理解 callback}
    \end{exampleblock}
  \end{columns}
\end{frame}

\imgslide{System：先看节点整体状态}{System.png}{%
  \begin{block}{先建立节点级视角}
    \begin{itemize}
      \item 当前 Erlang node 名称
      \item OTP / ERTS 版本
      \item 进程数量、端口、ETS、timer
      \item memory 和 scheduler 概览
    \end{itemize}
  \end{block}
  \begin{exampleblock}{讲课重点}
    BEAM VM 不是黑盒。Observer 直接展示节点内部资源。
  \end{exampleblock}
}

      \item Run Queue
      \item IO / Memory 趋势
    \end{itemize}
  \end{block}
  \begin{alertblock}{不要讲太深}
    入门阶段只需要知道：大量 Erlang process 由 BEAM scheduler 调度。
  \end{alertblock}
}

\imgslide{Supervisor Tree：OTP 结构可视化}{Applications_app_supervisor.png}{%
  \begin{block}{这张图最适合讲 OTP}
    \begin{itemize}
      \item supervisor 也是一个 Erlang process
      \item worker 是被 supervisor 管理的 process
      \item application 通过 supervision tree 组织运行时
    \end{itemize}
  \end{block}
  \begin{exampleblock}{关键结论}
    OTP 不是抽象概念，它在运行时就是一棵可观察的进程树。
  \end{exampleblock}
}

\imgslide{Messages / Dictionary / Stack Trace}{Process_Messages.png}{%

\begin{frame}{课堂演示顺序}
  \small
  \begin{enumerate}
    \item 启动 \code{rebar3 shell --sname application\_demo}
    \item 执行 \code{observer:start().}
    \item 看 \code{System}，说明 Observer 是节点观察工具
    \item 切到 \code{Applications}，选中 \code{full\_otp\_app}
    \item 展开 supervision tree
    \item 双击 \code{full\_otp\_app\_sup}，说明 supervisor 也是进程
    \item 双击 \code{kv\_server}，查看 Process Information 和 State
    \item 双击 \code{traffic\_light}，观察 gen\_statem 当前状态
    \item 打开 \code{Table Viewer}，查看 ETS / DETS
  \end{enumerate}
\end{frame}

\begin{frame}{小结}
  \begin{block}{本集结论}
    \begin{itemize}
      \item application 可以在 Observer 中看到
      \item supervisor tree 是运行时真实存在的进程树
      \item gen\_server 是有 mailbox 和 state 的 process
      \item gen\_statem 可以直接观察当前状态
      \item Table Viewer 可以辅助观察 ETS / DETS
    \end{itemize}
  \end{block}
  \begin{exampleblock}{下一步}
    在无 GUI、服务器、容器或 SSH 环境中，用 \code{observer\_cli} 获取类似的运行时视角。
  \end{exampleblock}
\end{frame}
