% ── Concepts ───────────────────────────────────────────────────────
\begin{frame}{Application 先分成两种}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{runtime application}
      \begin{itemize}
        \item 也叫 \keypoint{active application}
        \item 也叫 \keypoint{process-owning application}
        \item 启动后创建自己的 process
        \item 通常启动 top supervisor
        \item Observer 中能看到 supervision tree
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \footnotesize
    \begin{exampleblock}{library application}
      \begin{itemize}
        \item 也叫 \keypoint{passive application}
        \item 也叫 \keypoint{code-only application}
        \item 主要提供 module 和 function
        \item 通常没有自己的长期进程
        \item 下一集用 math 作为例子
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.25cm}
  \centering
  \good{两者都是 OTP application，区别在于是否拥有自己的运行时进程结构。}
\end{frame}



\begin{frame}[fragile]{这一集只抓一条主线：工具配置如何变成 OTP 入口}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{更自然的观察顺序}
\begin{lstlisting}
rebar.config
  -> project skeleton
      -> rebar3 compile
          -> application resource file
              -> mod callback
                  -> supervision tree
\end{lstlisting}
    \end{block}
    \column{0.48\textwidth}
    \footnotesize
    \begin{alertblock}{这一集的讲法}
      \begin{itemize}
        \item 先看 \code{rebar.config}：工具链把依赖和启动行为放在哪里。
        \item 再看初始化后的目录：此时还没有展开 \code{full\_otp\_app.app.src}，先理解工程骨架。
        \item 然后运行 \code{rebar3 compile}：下载依赖、分析工程、生成 application resource，再解释 OTP 入口。
      \end{itemize}
    \end{alertblock}
  \end{columns}
  \vspace{0.2cm}
  \centering
  \warn{目标不是从一堆字段开始背，而是顺着工具链看到 application 的入口自然出现。}
\end{frame}

\begin{frame}{配置很多，但先抓住三个有用字段}
  \begin{columns}[T,onlytextwidth]
    \column{0.31\textwidth}
    \begin{block}{\code{mod}}
      \footnotesize
      runtime application 的启动入口。它告诉 OTP：启动这个 app 时调用哪个 callback module。
      \vspace{0.15cm}
      \centering
      \good{最关键}
    \end{block}
    \column{0.31\textwidth}
    \begin{exampleblock}{\code{applications}}
      \footnotesize
      声明依赖哪些 OTP application，例如 \code{kernel}、\code{stdlib}、\code{observer\_cli}、\code{recon}。
      \vspace{0.15cm}
      \centering
      \good{看依赖}
    \end{exampleblock}
    \column{0.31\textwidth}
    \begin{alertblock}{\code{registered}}
      \footnotesize
      声明这个 app 可能注册的名字，方便工具、release 和读者理解系统边界。
      \vspace{0.15cm}
      \centering
      \good{看边界}
    \end{alertblock}
  \end{columns}
  \vspace{0.3cm}
  \centering
  \Large \keypoint{本集先不背字段清单，只用这些字段串起启动链路。}
\end{frame}

\begin{frame}[fragile]{\code{mod} 字段把配置交给 callback module}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
    \begin{block}{application callback}
\begin{lstlisting}
-module(full_otp_app_app).
-behaviour(application).

-export([start/2, stop/1]).

start(_StartType, _StartArgs) ->
    full_otp_app_sup:start_link().

stop(_State) ->
    ok.
\end{lstlisting}
    \end{block}
    \column{0.46\textwidth}
    \footnotesize
    \begin{exampleblock}{从配置到代码的关键跳转}
      \begin{itemize}
        \item \code{.app.src} 中的 \code{mod} 指向 \code{full\_otp\_app\_app}。
        \item OTP 启动 application 时调用 \code{start/2}。
        \item \code{start/2} 不承载业务逻辑，而是启动 top supervisor。
      \end{itemize}
    \end{exampleblock}
    \vspace{0.15cm}
    \begin{alertblock}{一句话}
      application 是启动边界；supervisor tree 才是系统长期运行的结构。
    \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{runtime application 不是 main 函数}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{block}{main 函数思维}
      \begin{itemize}
        \item 从入口执行一段逻辑
        \item 执行完可能就退出
        \item 重点是控制流
      \end{itemize}
    \end{block}
    \column{0.48\textwidth}
    \footnotesize
    \begin{exampleblock}{OTP application 思维}
      \begin{itemize}
        \item 从入口建立运行时结构
        \item 启动 top supervisor
        \item supervisor 管理长期 process
        \item 重点是生命周期和容错结构
      \end{itemize}
    \end{exampleblock}
  \end{columns}
  \vspace{0.25cm}
  \centering
  \good{runtime application 是启动边界，真正长期工作的是它启动出来的 supervisor 和 worker。}
\end{frame}

\begin{frame}{本集小结}

      \begin{itemize}
        \item 有 application resource file
        \item 通常有 \code{mod} 字段
        \item 通常有 application callback module
        \item \code{start/2} 通常启动 top supervisor
        \item 启动后拥有 supervision tree
      \end{itemize}

        \item \code{.app.src} 声明启动入口和依赖
        \item shell API 能 start / stop application
        \item Observer 能看到 app 的真实进程树
      \end{itemize}
    \end{exampleblock}
  \end{columns}
